cmake_minimum_required(VERSION 3.13)
project(wren_experiment C)
set(CMAKE_C_STANDARD 11)
include(wren.cmake)
add_executable(wren_host src/main.c src/experiment.c)
target_link_libraries(wren_host PRIVATE wren_vm m)
target_compile_options(wren_host PRIVATE -Wall -Wextra -Werror -O2)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME correctness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/test_host.py $<TARGET_FILE:wren_host>)
