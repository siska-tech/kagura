cmake_minimum_required(VERSION 3.13)
project(kagura_native_text C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(KJ_DESKTOP "Build native Desktop integration and regressions" ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/font_data.c
  COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_font_c.py ${CMAKE_CURRENT_BINARY_DIR}/font_data.c
  DEPENDS tools/gen_font_c.py ../../software/micropython/fonts/misaki8.kfnt)
add_library(kj_text STATIC src/kj_text.c ${CMAKE_CURRENT_BINARY_DIR}/font_data.c)
target_include_directories(kj_text PUBLIC include)
if(NOT MSVC)
  target_compile_options(kj_text PRIVATE -Wall -Wextra -Werror -Wconversion -fstack-usage)
endif()
enable_testing()
add_executable(kj_tests tests/test_text.c)
target_link_libraries(kj_tests PRIVATE kj_text)
add_test(NAME text_core COMMAND kj_tests)
add_executable(kj_oracle tests/oracle.c)
target_link_libraries(kj_oracle PRIVATE kj_text)
add_test(NAME unicode_font_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/oracle.py $<TARGET_FILE:kj_oracle>)
if(KJ_DESKTOP)
set(KD_SDL OFF CACHE BOOL "" FORCE)
add_subdirectory(../c-gem-desktop desktop)
set(variants kd_desktop_lib)
foreach(i RANGE 1 13)
  if(TARGET kd_desktop_lib_fault${i})
    list(APPEND variants kd_desktop_lib_fault${i})
  endif()
endforeach()
foreach(t IN LISTS variants)
  target_sources(${t} PRIVATE src/kj_desktop.c)
  target_include_directories(${t} PUBLIC include)
  target_compile_definitions(${t} PRIVATE KD_NATIVE_TEXT=1)
  target_link_libraries(${t} PUBLIC kj_text)
endforeach()
add_executable(kj_desktop_tests tests/test_desktop.c)
target_link_libraries(kj_desktop_tests PRIVATE kd_desktop_lib)
add_test(NAME text_desktop COMMAND kj_desktop_tests)
add_executable(kj_desktop_demo host/demo.c)
target_link_libraries(kj_desktop_demo PRIVATE kd_desktop_lib)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/demo)
add_test(NAME japanese_titles_demo COMMAND kj_desktop_demo ${CMAKE_CURRENT_BINARY_DIR}/demo)
endif()
