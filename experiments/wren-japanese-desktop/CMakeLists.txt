cmake_minimum_required(VERSION 3.13)
project(wren_japanese_desktop C)
set(CMAKE_C_STANDARD 11)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/gen_app.py RESULT_VARIABLE APP_STATUS)
if(NOT APP_STATUS EQUAL 0)
  message(FATAL_ERROR "Application generation failed")
endif()
include(integration.cmake)
add_executable(japanese_tests tests/main.c ../wren-desktop/tests/mock_hal.c)
target_include_directories(japanese_tests PRIVATE ../wren-desktop/tests)
target_link_libraries(japanese_tests PRIVATE integration)
target_compile_options(japanese_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME japanese_integration COMMAND japanese_tests)
add_executable(japanese_demo tests/demo.c ../wren-desktop/tests/mock_hal.c)
target_include_directories(japanese_demo PRIVATE ../wren-desktop/tests)
target_link_libraries(japanese_demo PRIVATE integration)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/wren-frames ${CMAKE_CURRENT_BINARY_DIR}/native-frames)
add_test(NAME japanese_wren_demo COMMAND japanese_demo wren ${CMAKE_CURRENT_BINARY_DIR}/wren-frames)
add_test(NAME japanese_native_demo COMMAND japanese_demo native ${CMAKE_CURRENT_BINARY_DIR}/native-frames)
