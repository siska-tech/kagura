cmake_minimum_required(VERSION 3.13)
if(STARFALL_PICO)
    include(../pico/pico_sdk_import.cmake)
endif()
project(project_starfall C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(starfall_logic game.c player.c bullet.c renderer.c enemy.c collision.c stage.c powerup.c assets.c music.c attract.c boss.c profile.c)
target_include_directories(starfall_logic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(starfall_logic PRIVATE -Wall -Wextra -Werror)
if(STARFALL_PICO)
    pico_sdk_init()
    option(STARFALL_QSPI "Use the QSPI FPGA bitstream" OFF)
    add_executable(starfall main.c input.c kagura_backend.c audio.c ../pico/kagura_host.c)
    pico_generate_pio_header(starfall ${CMAKE_CURRENT_LIST_DIR}/../pico/kagura_host.pio)
    target_include_directories(starfall PRIVATE ../pico ../include)
    target_compile_options(starfall PRIVATE -Wall -Wextra -Werror)
    target_compile_options(starfall PRIVATE -fstack-usage)
    target_compile_options(starfall_logic PRIVATE -fstack-usage)
    target_compile_definitions(starfall PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0 PICO_STACK_SIZE=0x1000)
    if(STARFALL_QSPI)
        target_compile_definitions(starfall PRIVATE STARFALL_LANES=KH_QSPI)
    endif()
    target_link_libraries(starfall starfall_logic pico_stdlib hardware_pio hardware_clocks)
    pico_enable_stdio_usb(starfall 1)
    pico_enable_stdio_uart(starfall 0)
    pico_add_extra_outputs(starfall)
else()
    enable_testing()
    add_executable(starfall_test tests/test_starfall.c)
    target_link_libraries(starfall_test starfall_logic)
    target_compile_options(starfall_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME deterministic_game COMMAND starfall_test)
    add_executable(starfall_flow_test tests/test_flow.c)
    target_link_libraries(starfall_flow_test starfall_logic)
    target_compile_options(starfall_flow_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME arcade_flow COMMAND starfall_flow_test)
    add_executable(starfall_boss_test tests/test_boss.c)
    target_link_libraries(starfall_boss_test starfall_logic)
    target_compile_options(starfall_boss_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME boss_phases COMMAND starfall_boss_test)
    add_executable(starfall_backend_test tests/test_backend.c kagura_backend.c audio.c)
    target_include_directories(starfall_backend_test PRIVATE tests ../pico ../include)
    target_link_libraries(starfall_backend_test starfall_logic)
    target_compile_options(starfall_backend_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME backend_contract COMMAND starfall_backend_test)
    add_executable(starfall_audio_test tests/test_audio.c audio.c)
    target_include_directories(starfall_audio_test PRIVATE ../pico ../include)
    target_link_libraries(starfall_audio_test starfall_logic)
    target_compile_options(starfall_audio_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME scene_music COMMAND starfall_audio_test)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME keyboard_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_keyboard.py)
    endif()
    add_executable(starfall_preview tests/preview.c)
    target_link_libraries(starfall_preview starfall_logic)
    target_compile_options(starfall_preview PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_executable(starfall_music_preview tests/music_preview.c)
    target_link_libraries(starfall_music_preview starfall_logic)
    target_compile_options(starfall_music_preview PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_executable(starfall_bench tests/bench.c)
    target_link_libraries(starfall_bench starfall_logic)
    target_compile_options(starfall_bench PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
