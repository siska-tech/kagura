# KAGURA audio HAL, KSNG song format and player: host build and tests.
#   cmake -S software/audio -B build/audio && cmake --build build/audio && ctest --test-dir build/audio
# The MicroPython module is built by the port (binding/); tools/test.sh runs everything.
cmake_minimum_required(VERSION 3.13)
project(kagura_audio C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
option(KA_SANITIZE "AddressSanitizer and UBSan" ON)
if(KA_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(ka_audio STATIC hal/ka_audio.c)
target_include_directories(ka_audio PUBLIC hal)

enable_testing()
add_executable(ka_hal_tests tests/test_hal.c)
target_include_directories(ka_hal_tests PRIVATE tests seq ${CMAKE_CURRENT_SOURCE_DIR}/../include)
target_link_libraries(ka_hal_tests ka_audio m)
add_test(NAME hal COMMAND ka_hal_tests)

# The song parser and the player, with mutants that the tests must catch.
foreach(fault 0 1 2 3 4)
    if(fault EQUAL 0)
        set(suffix "")
    else()
        set(suffix _fault${fault})
    endif()
    add_library(ks_seq${suffix} STATIC seq/ks_song.c seq/ks_player.c)
    target_include_directories(ks_seq${suffix} PUBLIC seq)
    target_link_libraries(ks_seq${suffix} PUBLIC ka_audio)
    target_compile_definitions(ks_seq${suffix} PRIVATE KS_FAULT=${fault})
    foreach(suite song player)
        add_executable(ks_${suite}_tests${suffix} tests/test_${suite}.c)
        target_include_directories(ks_${suite}_tests${suffix} PRIVATE tests songs)
        target_link_libraries(ks_${suite}_tests${suffix} ks_seq${suffix})
    endforeach()
    if(fault EQUAL 0)
        add_test(NAME song COMMAND ks_song_tests)
        add_test(NAME player COMMAND ks_player_tests)
    else()
        # Song faults 1..4, player faults 1..2: each must fail its suite.
        if(fault LESS_EQUAL 4)
            add_test(NAME song_fault${fault} COMMAND ks_song_tests${suffix})
            set_tests_properties(song_fault${fault} PROPERTIES WILL_FAIL TRUE)
        endif()
        if(fault LESS_EQUAL 2)
            add_test(NAME player_fault${fault} COMMAND ks_player_tests${suffix})
            set_tests_properties(player_fault${fault} PROPERTIES WILL_FAIL TRUE)
        endif()
    endif()
endforeach()
