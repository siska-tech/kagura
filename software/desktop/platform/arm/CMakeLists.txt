cmake_minimum_required(VERSION 3.13)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
set(CMAKE_C_FLAGS_INIT "-mcpu=cortex-m0plus -mthumb")
project(desktop_arm_cpu_tests C)
set(CMAKE_C_STANDARD 11)
include(../../cmake/modules.cmake)
target_compile_definitions(desktop_wren PRIVATE WD_MEMORY_DETAIL=1)
add_executable(integration_memory ${KD_WD}/tests/arm_host.c ${KD_WD}/tests/mock_hal.c ${KD_JP}/tests/main.c)
target_include_directories(integration_memory PRIVATE ${KD_WD}/tests)
target_compile_definitions(integration_memory PRIVATE WD_ARM_TESTS=1 main=integration_test_main)
target_compile_options(integration_memory PRIVATE -Os -ffunction-sections -fdata-sections -fstack-usage)
target_link_options(integration_memory PRIVATE -nostartfiles -static -Wl,--gc-sections)
target_link_libraries(integration_memory PRIVATE Kagura::WrenDesktop)
set_target_properties(integration_memory PROPERTIES SUFFIX .elf)
